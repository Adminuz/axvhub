02

range() generatori

start, stop va step parametrlari

range(start, stop, step) xaritasi

for sikli yozilishi

Rasmiy qo'llanmadagi 1.34-rasm sintaksisi

# 1. 0 dan 4 gacha chiqarish
for i in range(5):
    print(i, end=" ")  # 0 1 2 3 4

# 2. 1 dan 5 gacha chiqarish
for i in range(1, 6):
    print(i, end=" ")  # 1 2 3 4 5

# 3. 2 qadam bilan yurish
for i in range(0, 11, 2):
    print(i, end=" ")  # 0 2 4 6 8 10
03

Sonlar yig‘indisi

[1, N] intervaldagi sonlar yig‘indisini hisoblash

Akkumulyator sxemasi: s += i

[1, N] yig‘indisi kodi

Rasmiy qo'llanmadagi 1.40-rasm dasturi

n = int(input("N sonini kiriting: "))
s = 0

# n soni ham hisobga olinishi uchun n + 1 gacha
for i in range(1, n + 1):
    s += i

print(f"1 dan {n} gacha sonlar yig'indisi: {s}")
# Agar n = 10 bo'lsa, natija: 55

Ko‘paytirish jadvali dasturi

son = 7
print(f"--- {son} ning karra jadvali ---")

for i in range(1, 11):
    natija = son * i
    print(f"{son} x {i} = {natija}")
for i in range(1, 11): 10 ta qatordan iborat to'liq jadvalni bir zumda chizadi!

for siklida uchraydigan xatolar

Xato: print() ni sikl ichiga qo‘yish

# Xato: har safar chop etadi
for i in range(1, 6):
    s += i
    print(s) # 1, 3, 6, 10, 15!

# To'g'ri:
for i in range(1, 6):
    s += i
print(s)     # faqat 15
