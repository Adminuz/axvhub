3 ta urinishda tekshirish (1.44-rasm)

urinishlar = 0

while urinishlar < 3:
    parol = input("Parol kiriting: ")
    if parol == "1234":
        print("Kirish muvaffaqiyatli!")
        break  # Kod darhol to'xtaydi, sikl tugaydi
    urinishlar += 1

if urinishlar == 3:
    print("Urinishlar soni tugadi!")
