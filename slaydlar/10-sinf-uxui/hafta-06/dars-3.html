Sahifa maketi


body {
  display: grid;
  grid-template-columns: 220px 1fr;
  grid-template-areas:
    "header  header"
    "sidebar main"
    "footer  footer";
  min-height: 100vh;

header  { grid-area: header; }
aside   { grid-area: sidebar; }
main    { grid-area: main; }
