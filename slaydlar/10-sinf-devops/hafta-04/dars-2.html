#!/bin/bash
read -p "Son: " S
if [ "$S" -lt 0 ]; then
  echo "manfiy"
elif [ "$S" -eq 0 ]; then
  echo "nol"
else
  echo "musbat"
fi

[ ] va [[ ]]

[ ]

Hamma shellda ishlaydi. O'zgaruvchi atrofida tirnoq kerak. -a, -o o'rniga ikkita qavs.
VS

[[ ]]

Bash kengaytmasi. &&, || ichida ishlaydi, naqsh: == *.txt.

Pastga bosing: misol.

Qo'shimcha: naqsh bilan solishtirish

F="hisobot.txt"
[[ $F == *.txt ]] && echo "matn fayl"
[[ $F == a* || $F == h* ]] && echo "a yoki h"

Arifmetik amallar

Bash da hisob-kitob

Bo'lish butun songa bo'linadi

Natijani bosing.

Hisob-kitob kodi


A=17
B=5
echo $((A + B))      # 22
echo $((A / B))      # 3
echo $((A % B))      # 2
(( A % 2 == 0 )) && echo juft || echo toq
$(( )) ichida o'zgaruvchiga $ shart emas. (( )) shartda ishlatiladi.

Odatiy xatolar

  • [$A -eq 5]: qavs atrofida bo'sh joy yo'q.
  • [ $A -eq 5 ] da A bo'sh: tirnoq unutilgan.
  • Son uchun = yoki matn uchun -eq.
  • fi unutilgan yoki then yo'q.
  • $((7/2)) dan 3.5 kutish.

Tezkor bilim sinovi

echo $((17 % 5)) nima chiqaradi?

% bo'lishdan qoldiq: 17 = 3*5 + 2, qoldiq 2.
