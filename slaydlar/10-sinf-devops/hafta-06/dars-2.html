Versiyani oshirish


def bump(v, qism):
    major, minor, patch = map(int, v.split("."))
    if qism == "major":
        return f"{major + 1}.0.0"
    if qism == "minor":
        return f"{major}.{minor + 1}.0"
    return f"{major}.{minor}.{patch + 1}"

print(bump("2.4.1", "minor"))
MINOR oshganda PATCH 0 bo'ladi, MAJOR oshganda MINOR va PATCH 0 bo'ladi.
