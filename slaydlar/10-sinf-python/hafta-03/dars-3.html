class PostAPITestCase(APITestCase):
    def setUp(self):
        self.user = User.objects.create_user("u@mail.uz", "pass")
        self.cat = Category.objects.create(name="IT", slug="it")

    def test_list_posts(self):
        response = self.client.get("/api/v1/posts/")
        self.assertEqual(response.status_code, status.HTTP_200_OK)

    def test_create_post_authenticated(self):
        self.client.force_authenticate(user=self.user)
        res = self.client.post("/api/v1/posts/", {"title": "A", "category": self.cat.id})
        self.assertEqual(res.status_code, status.HTTP_201_CREATED)

    def test_create_unauthenticated_fails(self):
        res = self.client.post("/api/v1/posts/", {"title": "X"})
        self.assertEqual(res.status_code, status.HTTP_401_UNAUTHORIZED)
