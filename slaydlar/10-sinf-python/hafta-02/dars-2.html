class IsOwnerOrStaff(BasePermission):
    owner_field = "author"

    def has_object_permission(self, request, view, obj):
        if request.method in SAFE_METHODS:
            return True  # O'qishga hammaga ruxsat

        user = request.user
        if not (user and user.is_authenticated):
            return False
        if user.is_staff:
            return True  # Adminga to'liq ruxsat

        owner = getattr(obj, self.owner_field, None)
        return owner == user
03

perform_create amali

Muallifni server tomonida xavfsiz biriktirish

perform_create nima uchun kerak?

class PostViewSet(viewsets.ModelViewSet):
    queryset = Post.objects.all()
    lookup_field = "slug"
    permission_classes = (IsAuthenticatedOrReadOnly, IsOwnerOrStaff)

    def perform_create(self, serializer):
        # Muallifni frontenddan EMAS, tokendan olamiz!
        serializer.save(author=self.request.user)
Bu usul orqali hech kim boshqa odamning ID raqamini ko'rsatib, uning nomidan yolg'on maqola chop eta olmaydi.
