PostFilter va sana oraliqlari

src/apps/news/filters.py

import django_filters
from apps.news.models import Post

class PostFilter(django_filters.FilterSet):
    published_from = django_filters.IsoDateTimeFilter(
        field_name="published_at", lookup_expr="gte"
    )
    published_to = django_filters.IsoDateTimeFilter(
        field_name="published_at", lookup_expr="lte"
    )

    class Meta:
        model = Post
        fields = {
            "status": ["exact"],
            "category": ["exact"],
            "author": ["exact"],
            "tags": ["exact"],
        }

src/apps/news/views.py (PostViewSet)

class PostViewSet(viewsets.ModelViewSet):
    queryset = Post.objects.select_related("category", "author") \
                           .prefetch_related("tags").all()
    lookup_field = "slug"

    filterset_class = PostFilter

    search_fields = ("title", "excerpt", "body")

    ordering_fields = ("published_at", "created_at", "title", "status")
    ordering = ("-published_at", "-created_at")
