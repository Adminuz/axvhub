SELECT — asosiy sintaksis

-- Barcha ustunlar
SELECT * FROM yillik_kpi;

-- Aniq ustunlar
SELECT hudud_id, yil, oquvchilar_soni
FROM yillik_kpi;

-- Arifmetika va AS taxallus
SELECT
  hudud_id,
  yil,
  ROUND(oquvchilar_soni * 1.0
        / maktablar_soni, 1) AS maktab_yuklamasi
FROM yillik_kpi;

WHERE: solishtirish va mantiqiy operatorlar

Solishtirish

= != > < >= <=
WHERE yil = 2024
  AND oquvchilar_soni > 300000

Mantiqiy

AND  OR  NOT
WHERE yil = 2024
   OR yil = 2023
-- yoki qisqaroq:
WHERE yil IN (2023, 2024)

WHERE: BETWEEN, IN, LIKE

BETWEEN

WHERE yil
  BETWEEN 2022
  AND 2024

Ikkala chegara kiradi

IN

WHERE hudud_id
  IN (1, 3, 5)

OR zanjirini qisqartiradi

LIKE

WHERE nomi
  LIKE 'Toshkent%'

% = ixtiyoriy belgilar
_ = aynan 1 belgi

IS NULL — bo'sh qiymatlar

NOTO'G'RI

-- Bu DOIM FALSE qaytaradi!
WHERE turi = NULL

NULL == NULL — matematik jihatdan noto'g'ri. NULL = "noma'lum"

VS

TO'G'RI

-- Bo'sh qiymatlar
WHERE turi IS NULL

-- Bo'sh emas

Noma'lumni noma'lum bilan taqqoslab bo'lmaydi — IS NULL ishlatamiz!

ORDER BY — saralash

-- Bir ustun bo'yicha kamayish tartibi
SELECT hudud_id, yil, oquvchilar_soni
FROM yillik_kpi
WHERE yil = 2024
ORDER BY oquvchilar_soni DESC;

-- Ko'p ustunli saralash
SELECT hudud_id, yil, oquvchilar_soni
FROM yillik_kpi
ORDER BY yil DESC,        -- avval yangi yillar
         oquvchilar_soni DESC;  -- teng yillarda ko'proq birinchi

LIMIT va OFFSET — TOP-N va sahifalash

TOP-N tahlili

-- TOP-5 hudud (2024-yil)
SELECT h.nomi, k.oquvchilar_soni
FROM yillik_kpi k
JOIN hududlar h ON h.id = k.hudud_id
WHERE k.yil = 2024
ORDER BY k.oquvchilar_soni DESC
LIMIT 5;

Sahifalash (Pagination)

-- 1-sahifa (1-3 yozuv)
LIMIT 3 OFFSET 0

-- 2-sahifa (4-6 yozuv)
LIMIT 3 OFFSET 3

-- N-sahifa formulasi:
LIMIT size OFFSET (page-1)*size

Amaliyot: Python + sqlite3

import sqlite3

conn = sqlite3.connect("data/maktab.db")
conn.execute("PRAGMA foreign_keys = ON")
cur = conn.cursor()

# TOP-10 hudud — 2024-yil bo'yicha
sql = """
SELECT h.nomi, k.oquvchilar_soni
FROM yillik_kpi k
JOIN hududlar h ON h.id = k.hudud_id
WHERE k.yil = 2024
ORDER BY k.oquvchilar_soni DESC
LIMIT 10;
"""

cur.execute(sql)
print(f"{'Hudud':<25} {'O\'quvchilar soni':>18}")
print("-" * 45)
for nomi, son in cur.fetchall():
    print(f"{nomi:<25} {son:>18,}")

conn.close()

Murakkab filtr misoli

-- 2022-2024 da, o'quvchisi 400 000 dan ortiq hududlar,
-- maktab yuklamasi bilan, eng ko'p yuklamali birinchi
SELECT
  h.nomi                                          AS hudud,
  k.yil,
  k.oquvchilar_soni,
  ROUND(k.oquvchilar_soni * 1.0
        / k.maktablar_soni, 1)                   AS maktab_yuklamasi
FROM yillik_kpi k
JOIN hududlar h ON h.id = k.hudud_id
WHERE k.yil BETWEEN 2022 AND 2024
  AND k.oquvchilar_soni > 400000
ORDER BY k.yil DESC, maktab_yuklamasi DESC
LIMIT 6;

ORDER BY da alias (maktab_yuklamasi) ishlatish mumkin, lekin WHERE da emas — mantiqiy ijro tartibini eslang!
